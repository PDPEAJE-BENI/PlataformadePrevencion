// Cierra el menú al seleccionar una opción en móvil function cerrarMenu() { var menu = document.getElementById('menu-contenido'); if (window.innerWidth <= 600) { menu.style.display = 'none'; } } // Cerrar menú al hacer click fuera (solo móvil) document.addEventListener('click', function (e) { var menu = document.getElementById('menu-contenido'); var boton = document.getElementById('menu-boton'); if (window.innerWidth <= 600 && menu && boton) { if (!menu.contains(e.target) && !boton.contains(e.target)) { menu.style.display = 'none'; } } }); // Asegura que el menú se oculte al cambiar el tamaño de la ventana window.addEventListener('resize', function () { var menu = document.getElementById('menu-contenido'); if (window.innerWidth > 600) { menu.style.display = ''; } else { menu.style.display = 'none'; } }); // Corrige el cierre del menú al hacer scroll window.addEventListener('scroll', function () { var menu = document.getElementById('menu-contenido'); if (window.innerWidth <= 600 && menu) { menu.style.display = 'none'; } });
